cmake_minimum_required(VERSION 3.22.1)
project(vision_npu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LITERT_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/litert-2.3.0")
set(LITERT_BUILD_CONFIG_DISABLE_GPU 0)
set(LITERT_BUILD_CONFIG_DISABLE_NPU 0)
configure_file("${LITERT_HEADERS}/build_config.h.in"
    "${CMAKE_CURRENT_BINARY_DIR}/include/litert/build_common/build_config.h")
add_library(vision_npu SHARED npu_bridge.cpp)
target_include_directories(vision_npu PRIVATE "${LITERT_HEADERS}/include" "${CMAKE_CURRENT_BINARY_DIR}/include")
target_compile_options(vision_npu PRIVATE -Wall -Wextra -Werror)
target_link_options(vision_npu PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(vision_npu PRIVATE dl log)
